string TrimBoth(string s){ StringTrimLeft(s); StringTrimRight(s); return s; }

string TokenTail(string s)
{
   s = TrimBoth(s);
   int n = StringLen(s);
   if(n <= 8) return "***";
   return "..." + StringSubstr(s, n - 8);
}

string BaseUrlOnly(string url)
{
   string u = TrimBoth(url);
   int scheme = StringFind(u, "://");
   int start = (scheme >= 0 ? scheme + 3 : 0);
   int slash = StringFind(u, "/", start);
   if(slash > 0) return StringSubstr(u, 0, slash);
   return u;
}

string UrlNoQuery(string url)
{
   int q = StringFind(url, "?");
   if(q >= 0) return StringSubstr(url, 0, q);
   return url;
}

string RedactSensitiveUrl(string url)
{
   string out = url;
   int p = StringFind(out, "token=");
   if(p >= 0)
   {
      int e = StringFind(out, "&", p);
      if(e < 0) e = StringLen(out);
      out = StringSubstr(out, 0, p) + "token=***" + StringSubstr(out, e);
   }
   return out;
}

string JsonEscape(string s)
{
   string v=s;
   StringReplace(v,"\\","\\\\");
   StringReplace(v,"\"","\\\"");
   StringReplace(v,"\r"," ");
   StringReplace(v,"\n"," ");
   return v;
}

string UrlEncode(string s)
{
   string out="";
   for(int i=0;i<StringLen(s);i++)
   {
      ushort c=StringGetCharacter(s,i);
      if((c>='A'&&c<='Z')||(c>='a'&&c<='z')||(c>='0'&&c<='9')||c=='-'||c=='_'||c=='.'||c=='~')
         out += CharToString((uchar)c);
      else if(c==' ') out += "%20";
      else out += StringFormat("%%%02X",c);
   }
   return out;
}

bool HttpGet(string url, string &response, int &httpCode)
{
   char data[]; char result[]; string resultHeaders="";
   ArrayResize(data,0);
   ResetLastError();
   if(BEX_VERBOSE)
   {
      int qpos = StringFind(url, "?");
      string safePath = (qpos >= 0 ? StringSubstr(url, 0, qpos) : url);
      Print("BEX GET path=", safePath,
            " account=", AccountInfoInteger(ACCOUNT_LOGIN),
            " client=", InpClientId,
            " token_tail=", TokenTail(InpVipToken));
   }
   httpCode = WebRequest("GET", url, AuthHeaders(false), 8000, data, result, resultHeaders);
   if(httpCode == -1)
   {
      int err = GetLastError();
      Print("BEX HTTP GET failed err=",err," url=",RedactSensitiveUrl(UrlNoQuery(url))," full_redacted=",RedactSensitiveUrl(url));
      if(err == 4014) PrintWebRequestAllowlistHelp(url);
      response="";
      return false;
   }
   response = CharArrayToString(result,0,-1,CP_UTF8);
   if(BEX_VERBOSE) Print("BEX GET code=",httpCode," body=",response);
   return (httpCode>=200 && httpCode<300);
}

bool HttpPostJson(string url, string payload, string &response, int &httpCode)
{
   char data[]; char result[]; string resultHeaders="";
   int payloadLen = StringLen(payload);
   ArrayResize(data,0);
   if(payloadLen > 0) StringToCharArray(payload, data, 0, payloadLen, CP_UTF8);

   ResetLastError();
   if(BEX_VERBOSE)
      Print("BEX POST url=", url, " bytes=", payloadLen,
            " account=", AccountInfoInteger(ACCOUNT_LOGIN),
            " client=", InpClientId,
            " token_tail=", TokenTail(InpVipToken));

   httpCode = WebRequest("POST", url, AuthHeaders(true), 8000, data, result, resultHeaders);
   if(httpCode == -1)
   {
      int err = GetLastError();
      Print("BEX HTTP POST failed err=",err," url=",RedactSensitiveUrl(UrlNoQuery(url))," full_redacted=",RedactSensitiveUrl(url));
      if(err == 4014) PrintWebRequestAllowlistHelp(url);
      response="";
      return false;
   }
   response = CharArrayToString(result,0,-1,CP_UTF8);
   if(BEX_VERBOSE) Print("BEX POST code=",httpCode," body=",response);
   return (httpCode>=200 && httpCode<300);
}

bool ExtractJsonString(string json,string key,string &val)
{
   string pat="\""+key+"\"";
   int p=StringFind(json,pat); if(p<0) return false;
   int c=StringFind(json,":",p); if(c<0) return false;
   int q1=StringFind(json,"\"",c+1); if(q1<0) return false;
   int q2=q1+1;
   while(q2<StringLen(json))
   {
      int q=StringFind(json,"\"",q2); if(q<0) return false;
      if(StringGetCharacter(json,q-1)!='\\') { q2=q; break; }
      q2=q+1;
   }
   val=StringSubstr(json,q1+1,q2-q1-1);
   return true;
}

bool ExtractJsonNumber(string json,string key,double &val)
{
   string pat="\""+key+"\"";
   int p=StringFind(json,pat); if(p<0) return false;
   int c=StringFind(json,":",p); if(c<0) return false;
   int s=c+1;
   while(s<StringLen(json))
   {
      ushort ch=StringGetCharacter(json,s);
      if(ch!=' '&&ch!='\t'&&ch!='\r'&&ch!='\n') break;
      s++;
   }
   int e=s;
   while(e<StringLen(json))
   {
      ushort ch=StringGetCharacter(json,e);
      if(!((ch>='0'&&ch<='9')||ch=='.'||ch=='-'||ch=='+')) break;
      e++;
   }
   string n=TrimBoth(StringSubstr(json,s,e-s));
   if(n=="" || n=="null") return false;
   val=StringToDouble(n);
   return true;
}

bool ExtractJsonBool(string json,string key,bool &val)
{
   string pat="\""+key+"\"";
   int p=StringFind(json,pat); if(p<0) return false;
   int c=StringFind(json,":",p); if(c<0) return false;
   string tail=TrimBoth(StringSubstr(json,c+1,12));
   if(StringFind(tail,"true")==0){ val=true; return true; }
   if(StringFind(tail,"false")==0){ val=false; return true; }
   return false;
}

bool ExtractJsonStringAny(string json,string &out,string k1,string k2="",string k3="",string k4="")

int VolumeDigitsFromStep(double step)
{
   if(step<=0) return 2;
   if(step<0.001) return 4;
   if(step<0.01) return 3;
   return 2;
}

double NormalizeVolumeToBroker(string symbol,double volume)
{
   double minLot=SymbolInfoDouble(symbol,SYMBOL_VOLUME_MIN);
   double maxLot=SymbolInfoDouble(symbol,SYMBOL_VOLUME_MAX);
   double step=SymbolInfoDouble(symbol,SYMBOL_VOLUME_STEP);
   if(step<=0) step=0.01;
   volume=MathMax(volume,minLot);
   volume=MathMin(volume,maxLot);
   volume=MathMin(volume,InpMaxLot);
   volume=MathFloor(volume/step)*step;
   if(volume < minLot) volume = minLot;
   return NormalizeDouble(volume,VolumeDigitsFromStep(step));
}

double CurrentPrice(string symbol,string side)
{
   double bid=SymbolInfoDouble(symbol,SYMBOL_BID);
   double ask=SymbolInfoDouble(symbol,SYMBOL_ASK);
   side=Upper(side);
   if(side=="BUY") return ask;
   if(side=="SELL") return bid;
   if(BEX_USE_MID_PRICE && bid>0 && ask>0) return (bid+ask)/2.0;
   if(bid>0) return bid;
   return ask;
}

bool HasOpenPositionForSymbol(string symbol)
{
   for(int i=PositionsTotal()-1;i>=0;i--)
   {
      ulong t=PositionGetTicket(i); if(t==0) continue;
      if(!PositionSelectByTicket(t)) continue;
      if(PositionGetString(POSITION_SYMBOL)==symbol && PositionGetInteger(POSITION_MAGIC)==BEX_MAGIC_NUMBER) return true;
   }
   return false;
}

   }
   return false;
}

bool IsSafeBexShortComment(string comment)
{
   string c=TrimBoth(comment);
   if(StringFind(c,"BEX|")!=0) return false;
   if(StringLen(c)>31) return false;
   int pipes=0;
   for(int i=0;i<StringLen(c);i++) if(StringSubstr(c,i,1)=="|") pipes++;
   return (pipes==3);
}

string NormalizeEaOrderType(string orderType,string action,string side)
{
   string t=Upper(TrimBoth(orderType));
   string a=Upper(TrimBoth(action));
   string s=Upper(TrimBoth(side));

   if(t=="BUY" || t=="MARKET_BUY") return "BUY_MARKET";
   if(t=="SELL" || t=="MARKET_SELL") return "SELL_MARKET";
   if(t=="BUY_MARKET" || t=="SELL_MARKET" || t=="BUY_LIMIT" || t=="SELL_LIMIT" || t=="BUY_STOP" || t=="SELL_STOP") return t;

   if(StringFind(a,"BUY_LIMIT")>=0) return "BUY_LIMIT";
   if(StringFind(a,"SELL_LIMIT")>=0) return "SELL_LIMIT";
   if(StringFind(a,"BUY_STOP")>=0) return "BUY_STOP";
   if(StringFind(a,"SELL_STOP")>=0) return "SELL_STOP";
   if(StringFind(a,"BUY")>=0 && StringFind(a,"MARKET")>=0) return "BUY_MARKET";
   if(StringFind(a,"SELL")>=0 && StringFind(a,"MARKET")>=0) return "SELL_MARKET";

   if(s=="BUY") return "BUY_MARKET";
   if(s=="SELL") return "SELL_MARKET";
   return "NONE";
}

//============================= GUARD ENTRY ================================//
void TryExecuteFromGuard()
{
   if(!IsTokenReady()) return;

   string bexSymbol=GetBexSymbolToQuery();
   string brokerSymbol=GetBrokerSymbol(bexSymbol);
   if(!SymbolSelect(brokerSymbol,true)) { Print("BEX: cannot select symbol ",brokerSymbol); return; }

   double market=0.0;
   if(BEX_SEND_MARKET_PRICE) market=CurrentPrice(brokerSymbol,"");

   string url=BEX_GUARD_BASE_URL+BEX_GUARD_CHECK_PATH+"?symbol="+UrlEncode(bexSymbol);
   if(InpSendTokenInQuery) url += "&token="+UrlEncode(InpVipToken);
   url += "&client_id="+UrlEncode(InpClientId);
   url += "&account_login="+(string)AccountInfoInteger(ACCOUNT_LOGIN);
   url += "&account_server="+UrlEncode(AccountInfoString(ACCOUNT_SERVER));
   url += "&broker_symbol="+UrlEncode(brokerSymbol);
   url += "&magic="+(string)BEX_MAGIC_NUMBER;
   url += "&ea_contract=1";
   if(market>0) url += "&market_price="+DoubleToString(market,(int)SymbolInfoInteger(brokerSymbol,SYMBOL_DIGITS));

   string body=""; int code=0;
   if(!HttpGet(url,body,code)) return;

   bool ok=false, canExecute=false;
   if(!ExtractJsonBool(body,"ok",ok) || !ok) return;
   ExtractJsonBoolAny(body,canExecute,"ea_can_execute_now","can_execute_now","execute");

   string action="", orderType="", side="", fp="", sig="", setup="", planComment="", eaReason="";
   double lot=0, entry=0, sl=0, tp=0, conf=0, ageMs=0;

   ExtractJsonStringAny(body,action,"ea_action","action","decision");
   ExtractJsonStringAny(body,orderType,"ea_order_type","order_type","type");
   ExtractJsonStringAny(body,side,"ea_side","side","direction");
   ExtractJsonStringAny(body,fp,"fingerprint","signal_fingerprint","id");
   ExtractJsonStringAny(body,sig,"signal_id","id","signal_ref");
   ExtractJsonStringAny(body,setup,"setup_type","setup","selected_setup");
   ExtractJsonStringAny(body,planComment,"comment","mt5_comment","order_comment");
   ExtractJsonStringAny(body,eaReason,"ea_reason","reason","details_reason");

   ExtractJsonNumberAny(body,lot,"ea_lot","lot","volume");
   ExtractJsonNumberAny(body,entry,"ea_entry","entry","price");
   ExtractJsonNumberAny(body,sl,"ea_sl","sl","stop_loss");
   ExtractJsonNumberAny(body,tp,"ea_tp","tp","take_profit");
   ExtractJsonNumberAny(body,conf,"final_confidence","confidence","entry_confidence");
   ExtractJsonNumberAny(body,ageMs,"age_ms","signal_age_ms");

   side=Upper(TrimBoth(side));
   action=Upper(TrimBoth(action));
   orderType=NormalizeEaOrderType(orderType,action,side);
   if(sig=="") sig=fp;
   if(fp=="") fp=sig;
   if(fp=="") fp=bexSymbol+"|"+side+"|"+DoubleToString(entry,6)+"|"+DoubleToString(sl,6)+"|"+DoubleToString(tp,6);
   if(setup=="") setup="UNKNOWN";

   if(!canExecute || IsGuardWaitOrBlock(action,orderType))
   {
      if(BEX_VERBOSE)
         Print("BEX GUARD no executable trade: action=",action,
               " order_type=",orderType,
               " reason=",eaReason,
               " signal_id=",sig);
      return;
   }

   if(ageMs>0 && ageMs>(BEX_MAX_AGE_SECONDS*1000.0))
   {
      SendExecutionReport("BLOCKED_STALE",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Local stale block");
      return;
   }
   if(IsDuplicate(fp))
   {
      SendExecutionReport("BLOCKED_DUPLICATE",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Local duplicate block");
      return;
   }
   if(side!="BUY" && side!="SELL")
   {
      SendExecutionReport("BLOCKED_BAD_SIDE",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Bad side");
      return;
   }
   if(side=="BUY" && !BEX_ALLOW_BUY) return;
   if(side=="SELL" && !BEX_ALLOW_SELL) return;

   if(sl<=0 || tp<=0)
   {
      SendExecutionReport("BLOCKED_MISSING_SL_TP",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Missing SL/TP from Guard EA contract");
      return;
   }
   if(BEX_ONE_POSITION_ONLY && HasOpenPositionForSymbol(brokerSymbol))
   {
      SendExecutionReport("BLOCKED_OPEN_POSITION",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Open position exists");
      return;
   }
   if(BEX_SKIP_IF_PENDING && HasPendingOrderForSymbol(brokerSymbol))
   {
      SendExecutionReport("BLOCKED_PENDING_ORDER",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Pending order exists");
      return;
   }

   lot=NormalizeVolumeToBroker(brokerSymbol,lot);
   if(lot<=0)
   {
      SendExecutionReport("BLOCKED_BAD_LOT",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Bad lot");
      return;
   }

   if(!InpEnableAutoTrade)
   {
      SendExecutionReport("SIGNAL_ONLY",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"AutoTrade disabled");
      return;
   }

   int digits=(int)SymbolInfoInteger(brokerSymbol,SYMBOL_DIGITS);
   double bid=SymbolInfoDouble(brokerSymbol,SYMBOL_BID);
   double ask=SymbolInfoDouble(brokerSymbol,SYMBOL_ASK);

   if(entry<=0 && (orderType=="BUY_MARKET" || orderType=="SELL_MARKET")) entry=CurrentPrice(brokerSymbol,side);
   if(IsPendingOrderType(orderType) && entry<=0)
   {
      SendExecutionReport("BLOCKED_MISSING_ENTRY",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Pending order missing entry price");
      return;
   }

   // If a limit/stop trigger is already touched, execute market so EA does not miss the Guard signal.
   if(orderType=="BUY_LIMIT" && ask>0 && entry>=ask)  { orderType="BUY_MARKET";  entry=ask; }
   if(orderType=="SELL_LIMIT" && bid>0 && entry<=bid){ orderType="SELL_MARKET"; entry=bid; }
   if(orderType=="BUY_STOP" && ask>0 && entry<=ask)   { orderType="BUY_MARKET";  entry=ask; }
   if(orderType=="SELL_STOP" && bid>0 && entry>=bid) { orderType="SELL_MARKET"; entry=bid; }

   entry=NormalizeDouble(entry,digits);
   sl=NormalizeDouble(sl,digits);
   tp=NormalizeDouble(tp,digits);

   trade.SetExpertMagicNumber(BEX_MAGIC_NUMBER);
   trade.SetDeviationInPoints(InpSlippagePoints);
   string comment=IsSafeBexShortComment(planComment) ? planComment : MakeBexComment(setup,conf,bexSymbol);

   bool sent=false;
   if(orderType=="BUY_MARKET")       sent=trade.Buy(lot,brokerSymbol,0.0,sl,tp,comment);
   else if(orderType=="SELL_MARKET") sent=trade.Sell(lot,brokerSymbol,0.0,sl,tp,comment);
   else if(orderType=="BUY_LIMIT")   sent=trade.BuyLimit(lot,entry,brokerSymbol,sl,tp,ORDER_TIME_GTC,0,comment);
   else if(orderType=="SELL_LIMIT")  sent=trade.SellLimit(lot,entry,brokerSymbol,sl,tp,ORDER_TIME_GTC,0,comment);
   else if(orderType=="BUY_STOP")    sent=trade.BuyStop(lot,entry,brokerSymbol,sl,tp,ORDER_TIME_GTC,0,comment);
   else if(orderType=="SELL_STOP")   sent=trade.SellStop(lot,entry,brokerSymbol,sl,tp,ORDER_TIME_GTC,0,comment);
   else
   {
      SendExecutionReport("BLOCKED_UNSUPPORTED_ORDER_TYPE",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,0,sl,tp,0,"Unsupported EA order type: "+orderType);
      return;
   }

   long ret=trade.ResultRetcode();
   string msg=trade.ResultRetcodeDescription();
   double fill=trade.ResultPrice();

   if(sent)
   {
      MarkDuplicate(fp);
      string status=IsPendingOrderType(orderType) ? "PLACED_PENDING" : "FILLED";
      SendExecutionReport(status,bexSymbol,brokerSymbol,side,sig,fp,lot,entry,fill,sl,tp,ret,msg+" order_type="+orderType);
      Print("BEX TRADE SENT status=",status," order_type=",orderType,
            " symbol=",brokerSymbol," side=",side," lot=",lot,
            " entry=",entry," sl=",sl," tp=",tp," ret=",ret," ",msg);
   }
   else
   {
      SendExecutionReport("REJECTED",bexSymbol,brokerSymbol,side,sig,fp,lot,entry,fill,sl,tp,ret,msg+" order_type="+orderType+" bid="+DoubleToString(bid,digits)+" ask="+DoubleToString(ask,digits));
      Print("BEX TRADE REJECTED order_type=",orderType,
            " symbol=",brokerSymbol," side=",side," lot=",lot,
            " entry=",entry," sl=",sl," tp=",tp," ret=",ret," ",msg);
   }
}

int OnInit()
{
   Print("BEX VIP Client v1.16 Guard EA-contract + WebRequest 4014 fix started. Symbol=",_Symbol,
         " Account=",AccountInfoInteger(ACCOUNT_LOGIN),
         " client=",InpClientId,
         " token_tail=",TokenTail(InpVipToken));
   Print("BEX VIP: WebRequest allowlist must include EXACT base URLs: ", BaseUrlOnly(BEX_GUARD_BASE_URL), " and ", BaseUrlOnly(BEX_REPORT_BASE_URL));

   if(!IsTokenReady()) Print("BEX VIP: VipToken is empty.");
   trade.SetExpertMagicNumber(BEX_MAGIC_NUMBER);
   trade.SetDeviationInPoints(InpSlippagePoints);
   EventSetTimer(1);
   return INIT_SUCCEEDED;
}

void OnDeinit(const int reason)
{
   EventKillTimer();
   Comment("");
   Print("BEX VIP Client stopped. reason=",reason);
}

void OnTimer()
{
   datetime now=TimeCurrent();

   if(BEX_VERBOSE && (now-g_lastHeartbeat)>=10)
   {
      g_lastHeartbeat=now;
      Print("BEX TIMER heartbeat ", TimeToString(now),
            " account=", AccountInfoInteger(ACCOUNT_LOGIN),
            " symbol=", _Symbol,
            " client=", InpClientId,
            " token_tail=", TokenTail(InpVipToken));
   }

   if((now-g_lastPoll)>=BEX_POLL_SECONDS)
   {
      g_lastPoll=now;
      TryExecuteFromGuard();
   }

   Comment("BEX VIP Client v1.16 Guard Fixed\n",
           "Token: ",(StringLen(TrimBoth(InpVipToken))>0?"OK "+TokenTail(InpVipToken):"EMPTY"),"\n",
           "Client: ",InpClientId,"\n",
           "Account: ",AccountInfoInteger(ACCOUNT_LOGIN),"\n",
           "Chart Symbol: ",_Symbol,"\n",
           "BEX Symbol: ",GetBexSymbolToQuery(),"\n",
           "AutoTrade: ",(InpEnableAutoTrade?"ON":"OFF"),"\n",
           "Guard: ",BEX_GUARD_BASE_URL);
}
